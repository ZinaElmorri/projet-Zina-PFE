\chapter*{Liste des abréviations}
\addcontentsline{toc}{chapter}{Liste des abréviations}

\begin{description}[style=multiline,leftmargin=2.8cm,font=\bfseries]
    \item[API]    Interface de programmation d'applications (\textit{Application Programming Interface})
    \item[BIC]    Code d'identification bancaire (\textit{Bank Identifier Code})
    \item[EPC]    Conseil européen des paiements (\textit{European Payments Council})
    \item[HTTP]   Protocole de transfert hypertexte (\textit{Hypertext Transfer Protocol})
    \item[IA]     Intelligence artificielle
    \item[IBAN]   Numéro de compte bancaire international (\textit{International Bank Account Number})
    \item[ISO]    Organisation internationale de normalisation (\textit{International Organization for Standardization})
    \item[JPA]    Interface de persistance Java (\textit{Java Persistence API})
    \item[JSON]   Format de représentation de données (\textit{JavaScript Object Notation})
    \item[JWT]    Jeton web JSON (\textit{JSON Web Token})
    \item[KPI]    Indicateur clé de performance (\textit{Key Performance Indicator})
    \item[OTP]    Mot de passe à usage unique (\textit{One-Time Password})
    \item[PR-AUC] Aire sous la courbe précision--rappel (\textit{Precision--Recall Area Under Curve})
    \item[REST]   Transfert d'état de représentation (\textit{Representational State Transfer})
    \item[RH]     Ressources humaines
    \item[ROC-AUC] Aire sous la courbe ROC (\textit{Receiver Operating Characteristic})
    \item[SCT]    Virement SEPA (\textit{SEPA Credit Transfer})
    \item[SEPA]   Espace unique de paiements en euros (\textit{Single Euro Payments Area})
    \item[SIRH]   Système d'information des ressources humaines
    \item[SMTP]   Protocole de transfert de courriels (\textit{Simple Mail Transfer Protocol})
    \item[XML]    Langage de balisage extensible (\textit{Extensible Markup Language})
    \item[XSD]    Définition de schéma XML (\textit{XML Schema Definition})
\end{description}
